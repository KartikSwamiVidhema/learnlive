const mongoose = require('mongoose')
const Schema = mongoose.Schema

const blogSchema = new Schema(
  {
    name: {
      type: String,
      required: true
    },
    image: {
      type: String
    },
    description: {
      type: String
    },
    tags: {
      type: String
    },
    slug: {
      type: String,
    },
    excerpt :{
      type: String
    },
   date:{
      type: String
    },
   permalink:{
      type: String
    }
  },

  {
    timestamps: true
  }
);

module.exports = mongoose.model('blogs', blogSchema);
